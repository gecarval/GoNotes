// Lip Gloss example - borders gallery
//
// Example complexity rating: [★☆☆☆] 1/4
//
// Border sets the border shape (and optionally which sides to draw).
// BorderForeground colors it. Borders add to the rendered size, padding
// sits inside the border and margin outside.
package main

import (
	"charm.land/lipgloss/v2"
)

func main() {
	borders := []struct {
		name string
		b    lipgloss.Border
	}{
		{"Normal", lipgloss.NormalBorder()},
		{"Rounded", lipgloss.RoundedBorder()},
		{"Thick", lipgloss.ThickBorder()},
		{"Double", lipgloss.DoubleBorder()},
		{"Block", lipgloss.BlockBorder()},
		{"ASCII", lipgloss.ASCIIBorder()},
	}

	colors := []string{"#FF5F87", "#5FAFFF", "#87D787", "#FFD75F", "#AF87FF", "#D0D0D0"}

	var boxes []string
	for i, item := range borders {
		style := lipgloss.NewStyle().
			Border(item.b).
			BorderForeground(lipgloss.Color(colors[i])).
			Padding(0, 1).
			Width(13).
			Align(lipgloss.Center)
		boxes = append(boxes, style.Render(item.name))
	}

	// Three boxes per row.
	row1 := lipgloss.JoinHorizontal(lipgloss.Top, boxes[:3]...)
	row2 := lipgloss.JoinHorizontal(lipgloss.Top, boxes[3:]...)
	lipgloss.Println(lipgloss.JoinVertical(lipgloss.Left, row1, row2))

	// Only some sides: Border(b, top, right, bottom, left)
	underline := lipgloss.NewStyle().
		Border(lipgloss.NormalBorder(), false, false, true, false).
		BorderForeground(lipgloss.Color("#FF5F87"))
	lipgloss.Println(underline.Render("bottom border only"))
}
