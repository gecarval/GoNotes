// Bubbles example - list with the default delegate
//
// Example complexity rating: [★★☆☆] 2/4
//
// list.Model gives you navigation, pagination, fuzzy filtering (/) and a
// help line for free. Items implement list.Item (FilterValue); the
// DefaultDelegate also wants Title() and Description().
// The list needs a size: set it from tea.WindowSizeMsg.
//
// Keys: ↑/↓ move, / filter, enter choose, q quit
package main

import (
	"fmt"
	"os"

	"charm.land/bubbles/v2/list"
	tea "charm.land/bubbletea/v2"
	"charm.land/lipgloss/v2"
)

var docStyle = lipgloss.NewStyle().Margin(1, 2)

// book satisfies list.DefaultItem.
type book struct{ title, author string }

func (b book) Title() string       { return b.title }
func (b book) Description() string { return b.author }
func (b book) FilterValue() string { return b.title + " " + b.author }

type model struct {
	list   list.Model
	chosen string
}

func initialModel() model {
	items := []list.Item{
		book{"Dune", "Frank Herbert"},
		book{"Neuromancer", "William Gibson"},
		book{"The Left Hand of Darkness", "Ursula K. Le Guin"},
		book{"Foundation", "Isaac Asimov"},
		book{"Hyperion", "Dan Simmons"},
		book{"Snow Crash", "Neal Stephenson"},
		book{"Solaris", "Stanisław Lem"},
		book{"Blindsight", "Peter Watts"},
	}
	l := list.New(items, list.NewDefaultDelegate(), 0, 0) // size comes later
	l.Title = "Sci-fi shelf"
	l.SetStatusBarItemName("book", "books")
	return model{list: l}
}

func (m model) Init() tea.Cmd { return nil }

func (m model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.WindowSizeMsg:
		h, v := docStyle.GetFrameSize() // subtract the margins
		m.list.SetSize(msg.Width-h, msg.Height-v)
	case tea.KeyPressMsg:
		// While the user types a filter, keys belong to the filter input.
		if m.list.FilterState() == list.Filtering {
			break
		}
		if msg.String() == "enter" {
			if b, ok := m.list.SelectedItem().(book); ok {
				m.chosen = b.title
				return m, m.list.NewStatusMessage("Picked " + b.title)
			}
		}
	}
	var cmd tea.Cmd
	m.list, cmd = m.list.Update(msg) // the list handles q, ↑/↓, / ...
	return m, cmd
}

func (m model) View() tea.View {
	v := tea.NewView(docStyle.Render(m.list.View()))
	v.AltScreen = true
	return v
}

func main() {
	if _, err := tea.NewProgram(initialModel()).Run(); err != nil {
		fmt.Println("error:", err)
		os.Exit(1)
	}
}
