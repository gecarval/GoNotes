// Bubble Tea example - tick loop (countdown)
//
// Example complexity rating: [★★☆☆] 2/4
//
// tea.Tick fires ONE message after a duration. To get a repeating timer,
// return a new Tick from Update every time you receive the tick message.
// tea.Every is the same idea, but aligned with the system clock.
//
// Keys: space pause/resume, r reset, q quit
package main

import (
	"fmt"
	"os"
	"time"

	tea "charm.land/bubbletea/v2"
)

// tickMsg is our own message type. Any type can be a tea.Msg.
type tickMsg time.Time

// tick returns a command that sends tickMsg after one second.
func tick() tea.Cmd {
	return tea.Tick(time.Second, func(t time.Time) tea.Msg {
		return tickMsg(t)
	})
}

type model struct {
	remaining int
	paused    bool
}

func initialModel() model { return model{remaining: 10} }

func (m model) Init() tea.Cmd {
	return tick() // start the loop
}

func (m model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tickMsg:
		if m.paused || m.remaining == 0 {
			return m, nil // stop the loop by NOT returning another tick
		}
		m.remaining--
		return m, tick() // schedule the next tick
	case tea.KeyPressMsg:
		switch msg.String() {
		case "space":
			m.paused = !m.paused
			if !m.paused {
				return m, tick() // restart the loop
			}
		case "r":
			m.remaining = 10
		case "q", "ctrl+c":
			return m, tea.Quit
		}
	}
	return m, nil
}

func (m model) View() tea.View {
	state := "running"
	if m.paused {
		state = "paused"
	} else if m.remaining == 0 {
		state = "lift-off! 🚀"
	}
	return tea.NewView(fmt.Sprintf("T-minus %2d  (%s)\n\nspace pause • r reset • q quit\n", m.remaining, state))
}

func main() {
	if _, err := tea.NewProgram(initialModel()).Run(); err != nil {
		fmt.Println("error:", err)
		os.Exit(1)
	}
}
