// Bubble Tea example - run an external program ($EDITOR)
//
// Example complexity rating: [★★★☆] 3/4
//
// tea.ExecProcess pauses the program, gives the terminal to another
// interactive process (vim, nano, less...) and resumes when it exits.
// The callback turns the exit error into a message for Update.
//
// Keys: e open $EDITOR on a temp file, q quit
package main

import (
	"fmt"
	"os"
	"os/exec"

	tea "charm.land/bubbletea/v2"
)

type editorFinishedMsg struct {
	path string
	err  error
}

func openEditor(path string) tea.Cmd {
	editor := os.Getenv("EDITOR")
	if editor == "" {
		editor = "vi"
	}
	c := exec.Command(editor, path)
	return tea.ExecProcess(c, func(err error) tea.Msg {
		return editorFinishedMsg{path, err} // runs after the editor closes
	})
}

type model struct {
	content string
	err     error
}

func initialModel() model { return model{} }

func (m model) Init() tea.Cmd { return nil }

func (m model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case editorFinishedMsg:
		if msg.err != nil {
			m.err = msg.err
			return m, nil
		}
		data, err := os.ReadFile(msg.path)
		m.content, m.err = string(data), err
	case tea.KeyPressMsg:
		switch msg.String() {
		case "e":
			f, err := os.CreateTemp("", "scratch-*.txt")
			if err != nil {
				m.err = err
				return m, nil
			}
			f.WriteString(m.content)
			f.Close()
			return m, openEditor(f.Name())
		case "q", "ctrl+c":
			return m, tea.Quit
		}
	}
	return m, nil
}

func (m model) View() tea.View {
	if m.err != nil {
		return tea.NewView("error: " + m.err.Error() + "\n\ne try again • q quit\n")
	}
	body := m.content
	if body == "" {
		body = "(empty)"
	}
	return tea.NewView("Your text:\n\n" + body + "\n\ne edit in $EDITOR • q quit\n")
}

func main() {
	if _, err := tea.NewProgram(initialModel()).Run(); err != nil {
		fmt.Println("error:", err)
		os.Exit(1)
	}
}
