// Bubbles example - spinner while work runs in the background
//
// Example complexity rating: [★☆☆☆] 1/4
//
// A spinner animates by sending itself spinner.TickMsg. Start it from
// Init with m.spinner.Tick and keep forwarding TickMsg to its Update.
// tea.Batch starts the spinner AND the real work at the same time.
//
// Keys: s change spinner, q quit
package main

import (
	"fmt"
	"os"
	"time"

	"charm.land/bubbles/v2/spinner"
	tea "charm.land/bubbletea/v2"
	"charm.land/lipgloss/v2"
)

type doneMsg struct{}

func download() tea.Cmd {
	return func() tea.Msg {
		time.Sleep(4 * time.Second)
		return doneMsg{}
	}
}

var spinners = []spinner.Spinner{spinner.Dot, spinner.Line, spinner.MiniDot, spinner.Jump, spinner.Pulse, spinner.Points, spinner.Globe, spinner.Moon, spinner.Monkey}

type model struct {
	spinner spinner.Model
	kind    int
	done    bool
}

func initialModel() model {
	s := spinner.New(
		spinner.WithSpinner(spinner.Dot),
		spinner.WithStyle(lipgloss.NewStyle().Foreground(lipgloss.Color("205"))),
	)
	return model{spinner: s}
}

func (m model) Init() tea.Cmd {
	return tea.Batch(m.spinner.Tick, download()) // animate + work concurrently
}

func (m model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case doneMsg:
		m.done = true
		return m, nil // no more ticks: the spinner stops
	case spinner.TickMsg:
		var cmd tea.Cmd
		m.spinner, cmd = m.spinner.Update(msg)
		return m, cmd // returns the NEXT tick
	case tea.KeyPressMsg:
		switch msg.String() {
		case "s":
			m.kind = (m.kind + 1) % len(spinners)
			m.spinner.Spinner = spinners[m.kind]
		case "q", "ctrl+c":
			return m, tea.Quit
		}
	}
	return m, nil
}

func (m model) View() tea.View {
	if m.done {
		return tea.NewView("✔ Download complete!\n\nq quit\n")
	}
	return tea.NewView(fmt.Sprintf("%s Downloading assets...\n\ns change spinner • q quit\n", m.spinner.View()))
}

func main() {
	if _, err := tea.NewProgram(initialModel()).Run(); err != nil {
		fmt.Println("error:", err)
		os.Exit(1)
	}
}
