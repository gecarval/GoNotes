// Bubble Tea example - window size and alt screen
//
// Example complexity rating: [★☆☆☆] 1/4
//
// tea.WindowSizeMsg is delivered once at start-up and again on every
// terminal resize. Store it in the model so View can adapt to the size.
// Setting View.AltScreen draws on the full-screen alternate buffer.
//
// Keys: q quit (try resizing the terminal)
package main

import (
	"fmt"
	"os"
	"strings"

	tea "charm.land/bubbletea/v2"
)

type model struct {
	width, height int
}

func initialModel() model { return model{} }

func (m model) Init() tea.Cmd { return nil }

func (m model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.WindowSizeMsg:
		m.width, m.height = msg.Width, msg.Height // remember the size
	case tea.KeyPressMsg:
		if msg.String() == "q" || msg.String() == "ctrl+c" {
			return m, tea.Quit
		}
	}
	return m, nil
}

func (m model) View() tea.View {
	if m.width == 0 {
		return tea.NewView("waiting for size...") // first frame, before WindowSizeMsg
	}
	bar := strings.Repeat("═", m.width) // a line exactly as wide as the terminal
	body := fmt.Sprintf("%s\n\n  terminal is %d columns × %d rows\n\n  resize me, or press q\n", bar, m.width, m.height)

	v := tea.NewView(body)
	v.AltScreen = true // full-screen mode; the old screen comes back on exit
	v.WindowTitle = "window size demo"
	return v
}

func main() {
	if _, err := tea.NewProgram(initialModel()).Run(); err != nil {
		fmt.Println("error:", err)
		os.Exit(1)
	}
}
