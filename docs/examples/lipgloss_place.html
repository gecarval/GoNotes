Lip Gloss · centering a dialog with Place (inside Bubble Tea) ⭐⭐☆☆

lipgloss.Place puts a block inside a width×height area at a given position, filling the rest with whitespace. Combined with tea.WindowSizeMsg it keeps a dialog centered while the terminal resizes.

Keys: left/right choose, enter confirm, q quit

// Lip Gloss example - centering a dialog with Place (inside Bubble Tea)
//
// Example complexity rating: [★★☆☆] 2/4
//
// lipgloss.Place puts a block inside a width×height area at a given
// position, filling the rest with whitespace. Combined with
// tea.WindowSizeMsg it keeps a dialog centered while the terminal resizes.
//
// Keys: left/right choose, enter confirm, q quit
package main

import (
	"fmt"
	"os"

	tea "charm.land/bubbletea/v2"
	"charm.land/lipgloss/v2"
)

var (
	dialogStyle = lipgloss.NewStyle().
			Border(lipgloss.RoundedBorder()).
			BorderForeground(lipgloss.Color("#874BFD")).
			Padding(1, 3)
	buttonStyle = lipgloss.NewStyle().
			Foreground(lipgloss.Color("#FFF7DB")).
			Background(lipgloss.Color("#888B7E")).
			Padding(0, 3).
			MarginTop(1).MarginRight(2)
	activeButtonStyle = buttonStyle.
				Background(lipgloss.Color("#F25D94")).
				Underline(true)
)

type model struct {
	width, height int
	yes           bool
	answer        string
}

func initialModel() model { return model{yes: true} }

func (m model) Init() tea.Cmd { return nil }

func (m model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.WindowSizeMsg:
		m.width, m.height = msg.Width, msg.Height
	case tea.KeyPressMsg:
		switch msg.String() {
		case "left", "right", "tab":
			m.yes = !m.yes
		case "enter":
			m.answer = map[bool]string{true: "You said yes!", false: "Maybe next time."}[m.yes]
		case "q", "ctrl+c":
			return m, tea.Quit
		}
	}
	return m, nil
}

func (m model) View() tea.View {
	yes, no := buttonStyle, activeButtonStyle
	if m.yes {
		yes, no = activeButtonStyle, buttonStyle
	}
	buttons := lipgloss.JoinHorizontal(lipgloss.Top, yes.Render("Yes"), no.Render("No"))
	question := "Are you sure you want to eat marmalade?"
	if m.answer != "" {
		question = m.answer
	}
	dialog := dialogStyle.Render(lipgloss.JoinVertical(lipgloss.Center, question, buttons))

	// Center the dialog in the whole terminal, fill with a dotted pattern.
	screen := lipgloss.Place(m.width, m.height, lipgloss.Center, lipgloss.Center, dialog,
		lipgloss.WithWhitespaceChars("·"),
		lipgloss.WithWhitespaceStyle(lipgloss.NewStyle().Foreground(lipgloss.Color("#383838"))),
	)
	v := tea.NewView(screen)
	v.AltScreen = true
	return v
}

func main() {
	if _, err := tea.NewProgram(initialModel()).Run(); err != nil {
		fmt.Println("error:", err)
		os.Exit(1)
	}
}
