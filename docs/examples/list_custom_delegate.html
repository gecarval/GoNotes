// Bubbles example - list with a custom ItemDelegate
//
// Example complexity rating: [★★★☆] 3/4
//
// An ItemDelegate decides how ONE item is drawn (Render), how tall it is
// (Height), the gap between items (Spacing) and item-level keys (Update).
// Here: a compact one-line "todo" list where space toggles done.
//
// Keys: ↑/↓ move, space toggle, q quit
package main

import (
	"fmt"
	"io"
	"os"

	"charm.land/bubbles/v2/list"
	tea "charm.land/bubbletea/v2"
	"charm.land/lipgloss/v2"
)

var (
	itemStyle     = lipgloss.NewStyle().PaddingLeft(4)
	selectedStyle = lipgloss.NewStyle().PaddingLeft(2).Foreground(lipgloss.Color("#FF5F87"))
	doneStyle     = lipgloss.NewStyle().Strikethrough(true).Faint(true)
)

type todo struct {
	text string
	done bool
}

func (t todo) FilterValue() string { return t.text }

type delegate struct{}

func (d delegate) Height() int  { return 1 }
func (d delegate) Spacing() int { return 0 }

// Update handles keys that act on the selected item.
func (d delegate) Update(msg tea.Msg, m *list.Model) tea.Cmd {
	if key, ok := msg.(tea.KeyPressMsg); ok && key.String() == "space" {
		if t, ok := m.SelectedItem().(todo); ok {
			t.done = !t.done
			return m.SetItem(m.Index(), t) // replace the item in place
		}
	}
	return nil
}

// Render writes one item to w.
func (d delegate) Render(w io.Writer, m list.Model, index int, item list.Item) {
	t, ok := item.(todo)
	if !ok {
		return
	}
	box := "[ ] "
	text := t.text
	if t.done {
		box = "[x] "
		text = doneStyle.Render(text)
	}
	line := box + text
	if index == m.Index() {
		fmt.Fprint(w, selectedStyle.Render("> "+line))
		return
	}
	fmt.Fprint(w, itemStyle.Render(line))
}

type model struct{ list list.Model }

func initialModel() model {
	items := []list.Item{
		todo{"Buy coffee beans", true},
		todo{"Water the plants", false},
		todo{"Reply to emails", false},
		todo{"Book dentist appointment", false},
		todo{"Read one chapter", true},
	}
	l := list.New(items, delegate{}, 40, 12)
	l.Title = "Today"
	l.SetShowStatusBar(false)
	l.SetFilteringEnabled(false)
	return model{list: l}
}

func (m model) Init() tea.Cmd { return nil }

func (m model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	if ws, ok := msg.(tea.WindowSizeMsg); ok {
		m.list.SetWidth(ws.Width)
	}
	var cmd tea.Cmd
	m.list, cmd = m.list.Update(msg)
	return m, cmd
}

func (m model) View() tea.View { return tea.NewView("\n" + m.list.View()) }

func main() {
	if _, err := tea.NewProgram(initialModel()).Run(); err != nil {
		fmt.Println("error:", err)
		os.Exit(1)
	}
}
