// Bubble Tea example - basic counter
//
// Example complexity rating: [★☆☆☆] 1/4
//
// The smallest complete Bubble Tea program: a Model that holds a number,
// Update reacts to key presses and View renders the current state.
//
// Keys: up/k increment, down/j decrement, q quit
package main

import (
	"fmt"
	"os"

	tea "charm.land/bubbletea/v2"
)

// model is the whole state of the program.
type model struct {
	count int
}

func initialModel() model {
	return model{count: 0}
}

// Init runs once at start-up. Returning nil means "no initial command".
func (m model) Init() tea.Cmd {
	return nil
}

// Update receives every message (keys, resizes, your own msgs...) and
// returns the new model plus an optional command to run.
func (m model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.KeyPressMsg:
		switch msg.String() {
		case "up", "k":
			m.count++
		case "down", "j":
			m.count--
		case "q", "ctrl+c":
			return m, tea.Quit // tea.Quit is a Cmd: the program exits after this
		}
	}
	return m, nil
}

// View is a pure function of the model: same model, same screen.
func (m model) View() tea.View {
	s := fmt.Sprintf("Counter: %d\n\n↑/k increment • ↓/j decrement • q quit\n", m.count)
	return tea.NewView(s)
}

func main() {
	if _, err := tea.NewProgram(initialModel()).Run(); err != nil {
		fmt.Println("error:", err)
		os.Exit(1)
	}
}
