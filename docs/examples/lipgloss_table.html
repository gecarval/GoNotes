Lip Gloss · static table (lipgloss/table) ⭐⭐☆☆

lipgloss/table renders a table to a string. It is NOT interactive (for a scrollable, selectable table see bubbles/table). StyleFunc lets you style each cell based on its row and column.

// Lip Gloss example - static table (lipgloss/table)
//
// Example complexity rating: [★★☆☆] 2/4
//
// lipgloss/table renders a table to a string. It is NOT interactive
// (for a scrollable, selectable table see bubbles/table).
// StyleFunc lets you style each cell based on its row and column.
package main

import (
	"charm.land/lipgloss/v2"
	"charm.land/lipgloss/v2/table"
)

func main() {
	purple := lipgloss.Color("#7D56F4")
	headerStyle := lipgloss.NewStyle().Foreground(purple).Bold(true).Align(lipgloss.Center)
	cell := lipgloss.NewStyle().Padding(0, 1)
	odd := cell.Foreground(lipgloss.Color("#D0D0D0"))
	even := cell.Foreground(lipgloss.Color("#9E9E9E"))

	t := table.New().
		Border(lipgloss.NormalBorder()).
		BorderStyle(lipgloss.NewStyle().Foreground(purple)).
		Headers("LANGUAGE", "FORMAL", "INFORMAL").
		Row("Portuguese", "Olá, como está?", "Oi, tudo bem?").
		Row("English", "Hello, how are you?", "Hey, what's up?").
		Row("Spanish", "Hola, ¿cómo está?", "¿Qué tal?").
		Row("Japanese", "こんにちは", "やあ").
		StyleFunc(func(row, col int) lipgloss.Style {
			switch {
			case row == table.HeaderRow:
				return headerStyle
			case row%2 == 0:
				return even
			default:
				return odd
			}
		})

	lipgloss.Println(t) // *table.Table implements fmt.Stringer
}
