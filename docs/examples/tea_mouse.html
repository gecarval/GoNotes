// Bubble Tea example - mouse events
//
// Example complexity rating: [★★☆☆] 2/4
//
// Mouse reporting is enabled per frame through View.MouseMode.
// Clicks, releases, wheel and motion arrive as different message types,
// all implementing tea.MouseMsg (call .Mouse() to get X, Y and Button).
//
// Mouse: click anywhere, scroll the wheel. Keys: q quit
package main

import (
	"fmt"
	"os"

	tea "charm.land/bubbletea/v2"
)

type model struct {
	lastClick tea.Mouse
	clicks    int
	scroll    int
	hover     string
}

func initialModel() model { return model{} }

func (m model) Init() tea.Cmd { return nil }

func (m model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.MouseClickMsg:
		m.lastClick = msg.Mouse()
		m.clicks++
	case tea.MouseWheelMsg:
		switch msg.Mouse().Button {
		case tea.MouseWheelUp:
			m.scroll--
		case tea.MouseWheelDown:
			m.scroll++
		}
	case tea.MouseMotionMsg:
		mo := msg.Mouse()
		m.hover = fmt.Sprintf("(%d, %d)", mo.X, mo.Y)
	case tea.KeyPressMsg:
		if msg.String() == "q" || msg.String() == "ctrl+c" {
			return m, tea.Quit
		}
	}
	return m, nil
}

func (m model) View() tea.View {
	s := fmt.Sprintf(
		"clicks: %d   last: %s\nscroll: %d\nhover:  %s\n\nclick, scroll or move the mouse • q quit\n",
		m.clicks, m.lastClick, m.scroll, m.hover,
	)
	v := tea.NewView(s)
	v.MouseMode = tea.MouseModeAllMotion // or tea.MouseModeCellMotion (no plain hover)
	return v
}

func main() {
	if _, err := tea.NewProgram(initialModel()).Run(); err != nil {
		fmt.Println("error:", err)
		os.Exit(1)
	}
}
