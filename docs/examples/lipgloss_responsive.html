// Lip Gloss example - responsive two-column layout
//
// Example complexity rating: [★★★☆] 3/4
//
// To make boxes fill the terminal you must subtract the "frame" (border +
// padding + margin) from the available space. GetHorizontalFrameSize and
// GetVerticalFrameSize give you exactly that number.
//
// Keys: +/- change sidebar width, q quit (try resizing)
package main

import (
	"fmt"
	"os"
	"strings"

	tea "charm.land/bubbletea/v2"
	"charm.land/lipgloss/v2"
)

var (
	panel = lipgloss.NewStyle().
		Border(lipgloss.RoundedBorder()).
		BorderForeground(lipgloss.Color("#5FAFFF")).
		Padding(0, 1)
	header = lipgloss.NewStyle().Bold(true).Foreground(lipgloss.Color("#FF5F87"))
	footer = lipgloss.NewStyle().Faint(true)
)

type model struct {
	width, height int
	sidebar       int // sidebar width as a percentage
}

func initialModel() model { return model{sidebar: 30} }

func (m model) Init() tea.Cmd { return nil }

func (m model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.WindowSizeMsg:
		m.width, m.height = msg.Width, msg.Height
	case tea.KeyPressMsg:
		switch msg.String() {
		case "+", "=":
			m.sidebar = min(m.sidebar+5, 70)
		case "-":
			m.sidebar = max(m.sidebar-5, 15)
		case "q", "ctrl+c":
			return m, tea.Quit
		}
	}
	return m, nil
}

func (m model) View() tea.View {
	if m.width == 0 {
		return tea.NewView("")
	}
	top := header.Render("Responsive layout")
	bottom := footer.Render(fmt.Sprintf("%d×%d • +/- resize sidebar • q quit", m.width, m.height))

	// Space left for the panels after header and footer lines.
	bodyHeight := m.height - lipgloss.Height(top) - lipgloss.Height(bottom)

	// The frame (border+padding) is NOT part of Width/Height: subtract it.
	fw, fh := panel.GetHorizontalFrameSize(), panel.GetVerticalFrameSize()
	leftW := m.width * m.sidebar / 100
	rightW := m.width - leftW

	items := []string{"Inbox", "Drafts", "Sent", "Archive", "Trash"}
	left := panel.Width(leftW - fw).Height(bodyHeight - fh).Render(strings.Join(items, "\n"))
	right := panel.Width(rightW - fw).Height(bodyHeight - fh).
		Render("This column takes whatever space is left.\n\nResize the terminal: both panels follow.")

	body := lipgloss.JoinHorizontal(lipgloss.Top, left, right)
	v := tea.NewView(lipgloss.JoinVertical(lipgloss.Left, top, body, bottom))
	v.AltScreen = true
	return v
}

func main() {
	if _, err := tea.NewProgram(initialModel()).Run(); err != nil {
		fmt.Println("error:", err)
		os.Exit(1)
	}
}
