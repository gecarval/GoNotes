// Lip Gloss example - lists and trees (lipgloss/list, lipgloss/tree)
//
// Example complexity rating: [★☆☆☆] 1/4
//
// list and tree render nested structures with enumerators (bullets,
// numbers, roman numerals, tree branches). Lists can contain lists.
package main

import (
	"charm.land/lipgloss/v2"
	"charm.land/lipgloss/v2/list"
	"charm.land/lipgloss/v2/tree"
)

func main() {
	enum := lipgloss.NewStyle().Foreground(lipgloss.Color("#FF5F87")).MarginRight(1)

	groceries := list.New(
		"Fruit",
		list.New("Apples", "Oranges").Enumerator(list.Dash), // nested list
		"Bread",
		"Coffee",
	).Enumerator(list.Roman).EnumeratorStyle(enum)

	lipgloss.Println(groceries)
	lipgloss.Println()

	root := lipgloss.NewStyle().Bold(true).Foreground(lipgloss.Color("#5FAFFF"))
	project := tree.Root("my-app").
		Child(
			"go.mod",
			"main.go",
			tree.Root("ui").Child("model.go", "update.go", "view.go"),
			tree.Root("store").Child("sqlite.go"),
		).
		Enumerator(tree.RoundedEnumerator).
		RootStyle(root).
		EnumeratorStyle(lipgloss.NewStyle().Foreground(lipgloss.Color("#6C6C6C")))

	lipgloss.Println(project)
}
