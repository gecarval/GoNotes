// Bubbles example - multi-line textarea
//
// Example complexity rating: [★★☆☆] 2/4
//
// textarea is the multi-line sibling of textinput: same Focus/Blur/Value
// API, plus SetWidth/SetHeight, line numbers and a character limit.
// Resize it from tea.WindowSizeMsg so it fills the terminal.
//
// Keys: type, esc toggle focus, ctrl+s "save", ctrl+c quit
package main

import (
	"fmt"
	"os"

	"charm.land/bubbles/v2/textarea"
	tea "charm.land/bubbletea/v2"
	"charm.land/lipgloss/v2"
)

var status = lipgloss.NewStyle().Faint(true)

type model struct {
	ta    textarea.Model
	saved string
}

func initialModel() model {
	ta := textarea.New()
	ta.Placeholder = "Once upon a time..."
	ta.ShowLineNumbers = true
	ta.CharLimit = 2000
	ta.SetWidth(50)
	ta.SetHeight(8)
	ta.Focus()
	return model{ta: ta}
}

func (m model) Init() tea.Cmd { return textarea.Blink }

func (m model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.WindowSizeMsg:
		m.ta.SetWidth(msg.Width)
		m.ta.SetHeight(msg.Height - 4) // leave room for title + status lines
	case tea.KeyPressMsg:
		switch msg.String() {
		case "esc":
			if m.ta.Focused() {
				m.ta.Blur()
			} else {
				return m, m.ta.Focus()
			}
		case "ctrl+s":
			m.saved = fmt.Sprintf("saved %d chars, %d lines", m.ta.Length(), m.ta.LineCount())
			return m, nil
		case "ctrl+c":
			return m, tea.Quit
		}
	}
	var cmd tea.Cmd
	m.ta, cmd = m.ta.Update(msg)
	return m, cmd
}

func (m model) View() tea.View {
	info := fmt.Sprintf("line %d, col %d • esc focus • ctrl+s save • ctrl+c quit  %s",
		m.ta.Line()+1, m.ta.Column()+1, m.saved)
	return tea.NewView("Tell me a story.\n\n" + m.ta.View() + "\n" + status.Render(info))
}

func main() {
	if _, err := tea.NewProgram(initialModel()).Run(); err != nil {
		fmt.Println("error:", err)
		os.Exit(1)
	}
}
