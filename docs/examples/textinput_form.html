// Bubbles example - multi-field form with focus cycling
//
// Example complexity rating: [★★★☆] 3/4
//
// Several inputs, only one focused at a time. Moving focus means Blur()
// on the old one and Focus() on the new one. Each input's Update returns
// a Cmd, so collect them all and return them with tea.Batch.
// EchoPassword hides the typed characters.
//
// Keys: tab/shift+tab move, enter next/submit, esc quit
package main

import (
	"fmt"
	"os"
	"strings"

	"charm.land/bubbles/v2/textinput"
	tea "charm.land/bubbletea/v2"
	"charm.land/lipgloss/v2"
)

var (
	focused = lipgloss.NewStyle().Foreground(lipgloss.Color("#FF5F87"))
	blurred = lipgloss.NewStyle().Foreground(lipgloss.Color("#6C6C6C"))
)

type model struct {
	inputs    []textinput.Model
	focus     int
	submitted bool
}

func initialModel() model {
	labels := []string{"Name", "Email", "Password"}
	m := model{inputs: make([]textinput.Model, len(labels))}
	for i, label := range labels {
		t := textinput.New()
		t.Placeholder = label
		t.SetWidth(30)
		if label == "Password" {
			t.EchoMode = textinput.EchoPassword
			t.EchoCharacter = '•'
		}
		m.inputs[i] = t
	}
	m.inputs[0].Focus()
	return m
}

func (m model) Init() tea.Cmd { return textinput.Blink }

// setFocus blurs every input except i.
func (m *model) setFocus(i int) tea.Cmd {
	m.focus = (i + len(m.inputs)) % len(m.inputs) // wrap around
	var cmd tea.Cmd
	for j := range m.inputs {
		if j == m.focus {
			cmd = m.inputs[j].Focus()
		} else {
			m.inputs[j].Blur()
		}
	}
	return cmd
}

func (m model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	if key, ok := msg.(tea.KeyPressMsg); ok {
		switch key.String() {
		case "esc", "ctrl+c":
			return m, tea.Quit
		case "tab", "down":
			return m, m.setFocus(m.focus + 1)
		case "shift+tab", "up":
			return m, m.setFocus(m.focus - 1)
		case "enter":
			if m.focus == len(m.inputs)-1 {
				m.submitted = true
				return m, nil
			}
			return m, m.setFocus(m.focus + 1)
		}
	}

	// Update all inputs (only the focused one reacts to keys).
	cmds := make([]tea.Cmd, len(m.inputs))
	for i := range m.inputs {
		m.inputs[i], cmds[i] = m.inputs[i].Update(msg)
	}
	return m, tea.Batch(cmds...)
}

func (m model) View() tea.View {
	var b strings.Builder
	b.WriteString("Create account\n\n")
	for i, in := range m.inputs {
		style := blurred
		if i == m.focus {
			style = focused
		}
		b.WriteString(style.Render(in.Placeholder+":") + "\n" + in.View() + "\n\n")
	}
	if m.submitted {
		b.WriteString(focused.Render(fmt.Sprintf("Welcome, %s!", m.inputs[0].Value())) + "\n\n")
	}
	b.WriteString(blurred.Render("tab/shift+tab move • enter next • esc quit"))
	return tea.NewView(b.String())
}

func main() {
	if _, err := tea.NewProgram(initialModel()).Run(); err != nil {
		fmt.Println("error:", err)
		os.Exit(1)
	}
}
