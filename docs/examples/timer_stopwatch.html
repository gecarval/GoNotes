// Bubbles example - timer and stopwatch side by side
//
// Example complexity rating: [★★☆☆] 2/4
//
// timer counts down and sends timer.TimeoutMsg at zero; stopwatch counts
// up. Both tick through their own TickMsg; each message carries an ID,
// so forwarding a msg to BOTH components is safe: the wrong one ignores it.
//
// Keys: space start/stop both, r reset stopwatch, q quit
package main

import (
	"fmt"
	"os"
	"time"

	"charm.land/bubbles/v2/stopwatch"
	"charm.land/bubbles/v2/timer"
	tea "charm.land/bubbletea/v2"
	"charm.land/lipgloss/v2"
)

var box = lipgloss.NewStyle().Border(lipgloss.RoundedBorder()).Padding(0, 2).Width(22)

type model struct {
	timer     timer.Model
	stopwatch stopwatch.Model
	timedOut  bool
}

func initialModel() model {
	return model{
		timer:     timer.New(30*time.Second, timer.WithInterval(100*time.Millisecond)),
		stopwatch: stopwatch.New(stopwatch.WithInterval(100 * time.Millisecond)),
	}
}

func (m model) Init() tea.Cmd {
	return tea.Batch(m.timer.Init(), m.stopwatch.Init()) // start both
}

func (m model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case timer.TimeoutMsg:
		m.timedOut = true
	case tea.KeyPressMsg:
		switch msg.String() {
		case "space":
			return m, tea.Batch(m.timer.Toggle(), m.stopwatch.Toggle())
		case "r":
			return m, m.stopwatch.Reset()
		case "q", "ctrl+c":
			return m, tea.Quit
		}
	}
	// Forward to both: TickMsg/StartStopMsg carry the component ID.
	var c1, c2 tea.Cmd
	m.timer, c1 = m.timer.Update(msg)
	m.stopwatch, c2 = m.stopwatch.Update(msg)
	return m, tea.Batch(c1, c2)
}

func (m model) View() tea.View {
	t := "timer\n" + m.timer.View()
	if m.timedOut {
		t = "timer\nTime's up!"
	}
	s := "stopwatch\n" + m.stopwatch.View()
	row := lipgloss.JoinHorizontal(lipgloss.Top, box.Render(t), box.Render(s))
	return tea.NewView(row + "\n space start/stop • r reset stopwatch • q quit\n")
}

func main() {
	if _, err := tea.NewProgram(initialModel()).Run(); err != nil {
		fmt.Println("error:", err)
		os.Exit(1)
	}
}
