// Bubble Tea example - send messages from outside (Program.Send)
//
// Example complexity rating: [★★★☆] 3/4
//
// Sometimes events come from somewhere Bubble Tea doesn't control: a
// websocket, a file watcher, another goroutine. Keep the *tea.Program and
// call p.Send(msg): the message goes through Update like any other.
//
// Keys: q quit
package main

import (
	"fmt"
	"math/rand/v2"
	"os"
	"time"

	tea "charm.land/bubbletea/v2"
)

type logLineMsg string

type model struct {
	lines []string
}

func initialModel() model { return model{} }

func (m model) Init() tea.Cmd { return nil }

func (m model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case logLineMsg:
		m.lines = append(m.lines, string(msg))
		if len(m.lines) > 8 {
			m.lines = m.lines[1:] // keep only the last 8 lines
		}
	case tea.KeyPressMsg:
		if msg.String() == "q" || msg.String() == "ctrl+c" {
			return m, tea.Quit
		}
	}
	return m, nil
}

func (m model) View() tea.View {
	s := "Live log (from another goroutine):\n\n"
	for _, l := range m.lines {
		s += "  " + l + "\n"
	}
	return tea.NewView(s + "\nq quit\n")
}

func main() {
	p := tea.NewProgram(initialModel())

	// A producer that knows nothing about the model: it only sends messages.
	go func() {
		levels := []string{"INFO", "WARN", "DEBUG"}
		for i := 1; ; i++ {
			time.Sleep(time.Duration(200+rand.IntN(600)) * time.Millisecond)
			p.Send(logLineMsg(fmt.Sprintf("[%s] event #%d", levels[rand.IntN(3)], i)))
		}
	}()

	if _, err := p.Run(); err != nil {
		fmt.Println("error:", err)
		os.Exit(1)
	}
}
