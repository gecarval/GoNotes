// Bubbles example - viewport pager with header and footer
//
// Example complexity rating: [★★☆☆] 2/4
//
// viewport shows a window into content taller than the screen and
// handles scrolling keys and the mouse wheel. Its height must be the
// terminal height MINUS whatever you draw around it (header/footer).
//
// Keys: ↑/↓/pgup/pgdn/mouse wheel scroll, q quit
package main

import (
	"fmt"
	"os"
	"strings"

	"charm.land/bubbles/v2/viewport"
	tea "charm.land/bubbletea/v2"
	"charm.land/lipgloss/v2"
)

var (
	titleStyle = lipgloss.NewStyle().Bold(true).Padding(0, 1).
			Background(lipgloss.Color("#7D56F4")).Foreground(lipgloss.Color("#FFFFFF"))
	infoStyle = lipgloss.NewStyle().Foreground(lipgloss.Color("#7D56F4"))
)

type model struct {
	content string
	ready   bool
	vp      viewport.Model
}

func initialModel() model {
	var b strings.Builder
	for i := 1; i <= 120; i++ {
		fmt.Fprintf(&b, "%3d  The quick brown fox jumps over the lazy dog.\n", i)
	}
	return model{content: b.String()}
}

func (m model) Init() tea.Cmd { return nil }

func (m model) headerView() string { return titleStyle.Render("fox.txt") }

func (m model) footerView() string {
	return infoStyle.Render(fmt.Sprintf("%3.f%%", m.vp.ScrollPercent()*100))
}

func (m model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.KeyPressMsg:
		if s := msg.String(); s == "q" || s == "ctrl+c" {
			return m, tea.Quit
		}
	case tea.WindowSizeMsg:
		margin := lipgloss.Height(m.headerView()) + lipgloss.Height(m.footerView())
		if !m.ready {
			// The first WindowSizeMsg: now we know the size, create it.
			m.vp = viewport.New(viewport.WithWidth(msg.Width), viewport.WithHeight(msg.Height-margin))
			m.vp.SetContent(m.content)
			m.ready = true
		} else {
			m.vp.SetWidth(msg.Width)
			m.vp.SetHeight(msg.Height - margin)
		}
	}
	var cmd tea.Cmd
	m.vp, cmd = m.vp.Update(msg)
	return m, cmd
}

func (m model) View() tea.View {
	if !m.ready {
		return tea.NewView("\n  Initializing...")
	}
	v := tea.NewView(m.headerView() + "\n" + m.vp.View() + "\n" + m.footerView())
	v.AltScreen = true
	v.MouseMode = tea.MouseModeCellMotion // needed for the mouse wheel
	return v
}

func main() {
	if _, err := tea.NewProgram(initialModel()).Run(); err != nil {
		fmt.Println("error:", err)
		os.Exit(1)
	}
}
