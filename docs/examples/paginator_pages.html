// Bubbles example - paginating a long slice
//
// Example complexity rating: [★★☆☆] 2/4
//
// paginator only does the math and the "• ○ ○" indicator: you tell it
// how many items exist (SetTotalPages) and ask GetSliceBounds which
// part of YOUR slice belongs on the current page.
//
// Keys: ←/→ (h/l) change page, q quit
package main

import (
	"fmt"
	"os"
	"strings"

	"charm.land/bubbles/v2/paginator"
	tea "charm.land/bubbletea/v2"
	"charm.land/lipgloss/v2"
)

type model struct {
	items []string
	pages paginator.Model
}

func initialModel() model {
	var items []string
	for i := 1; i <= 42; i++ {
		items = append(items, fmt.Sprintf("Item %02d", i))
	}
	p := paginator.New()
	p.Type = paginator.Dots // or paginator.Arabic → "2/5"
	p.PerPage = 8
	p.ActiveDot = lipgloss.NewStyle().Foreground(lipgloss.Color("#FF5F87")).Render("•")
	p.InactiveDot = lipgloss.NewStyle().Foreground(lipgloss.Color("#4E4E4E")).Render("•")
	p.SetTotalPages(len(items))
	return model{items: items, pages: p}
}

func (m model) Init() tea.Cmd { return nil }

func (m model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	if key, ok := msg.(tea.KeyPressMsg); ok {
		if s := key.String(); s == "q" || s == "ctrl+c" {
			return m, tea.Quit
		}
	}
	var cmd tea.Cmd
	m.pages, cmd = m.pages.Update(msg) // handles ←/→, h/l, pgup/pgdn
	return m, cmd
}

func (m model) View() tea.View {
	var b strings.Builder
	b.WriteString("\n  Paginator\n\n")
	start, end := m.pages.GetSliceBounds(len(m.items))
	for _, item := range m.items[start:end] {
		b.WriteString("  • " + item + "\n")
	}
	b.WriteString("\n  " + m.pages.View() + "\n\n  h/l ←/→ page • q quit\n")
	return tea.NewView(b.String())
}

func main() {
	if _, err := tea.NewProgram(initialModel()).Run(); err != nil {
		fmt.Println("error:", err)
		os.Exit(1)
	}
}
