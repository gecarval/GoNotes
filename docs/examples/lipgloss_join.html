Lip Gloss · JoinHorizontal / JoinVertical layout ⭐⭐☆☆

Layout in Lip Gloss is string composition: render each block, then glue blocks side by side (JoinHorizontal) or stacked (JoinVertical). The Position argument aligns blocks of different heights/widths.

// Lip Gloss example - JoinHorizontal / JoinVertical layout
//
// Example complexity rating: [★★☆☆] 2/4
//
// Layout in Lip Gloss is string composition: render each block, then glue
// blocks side by side (JoinHorizontal) or stacked (JoinVertical).
// The Position argument aligns blocks of different heights/widths.
package main

import (
	"charm.land/lipgloss/v2"
)

func main() {
	box := lipgloss.NewStyle().
		Border(lipgloss.RoundedBorder()).
		BorderForeground(lipgloss.Color("#7D56F4")).
		Padding(0, 1)

	short := box.Render("short\nblock")
	tall := box.Render("a\ntaller\nblock\nwith\nlines")
	wide := box.Width(40).Render("a wide block below the others")

	// Side by side, aligned at the TOP...
	top := lipgloss.JoinHorizontal(lipgloss.Top, short, tall, short)
	// ...or at the BOTTOM / CENTER (0 = top, 0.5 = center, 1 = bottom).
	bottom := lipgloss.JoinHorizontal(lipgloss.Bottom, short, tall, short)

	label := lipgloss.NewStyle().Faint(true)

	// Stack everything vertically, left-aligned.
	page := lipgloss.JoinVertical(lipgloss.Left,
		label.Render("JoinHorizontal(Top, ...)"), top,
		label.Render("JoinHorizontal(Bottom, ...)"), bottom,
		wide,
	)
	lipgloss.Println(page)

	// Width/Height measure rendered strings (ANSI-aware).
	w, h := lipgloss.Size(page)
	lipgloss.Printf("page is %d×%d cells\n", w, h)
}
