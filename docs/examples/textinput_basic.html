// Bubbles example - single text input
//
// Example complexity rating: [★☆☆☆] 1/4
//
// Every Bubbles component is a small Bubble Tea model: keep it inside
// your model, forward messages to its Update, and print its View.
// Focus() returns a Cmd (cursor blink) and the input only reacts to keys
// while focused.
//
// Keys: type, enter submit, esc quit
package main

import (
	"fmt"
	"os"

	"charm.land/bubbles/v2/textinput"
	tea "charm.land/bubbletea/v2"
)

type model struct {
	input  textinput.Model
	answer string
}

func initialModel() model {
	ti := textinput.New()
	ti.Placeholder = "Pikachu"
	ti.Prompt = "› "
	ti.CharLimit = 32
	ti.SetWidth(24)
	ti.Focus() // without focus the input ignores key presses
	return model{input: ti}
}

func (m model) Init() tea.Cmd {
	return textinput.Blink // start the cursor blinking
}

func (m model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	if key, ok := msg.(tea.KeyPressMsg); ok {
		switch key.String() {
		case "enter":
			m.answer = m.input.Value()
			m.input.Reset()
			return m, nil
		case "esc", "ctrl+c":
			return m, tea.Quit
		}
	}

	// Forward EVERY other message to the component, then keep its new state.
	var cmd tea.Cmd
	m.input, cmd = m.input.Update(msg)
	return m, cmd
}

func (m model) View() tea.View {
	s := "What's your favorite Pokémon?\n\n" + m.input.View() + "\n\n"
	if m.answer != "" {
		s += fmt.Sprintf("You chose %q!\n\n", m.answer)
	}
	return tea.NewView(s + "enter submit • esc quit\n")
}

func main() {
	if _, err := tea.NewProgram(initialModel()).Run(); err != nil {
		fmt.Println("error:", err)
		os.Exit(1)
	}
}
