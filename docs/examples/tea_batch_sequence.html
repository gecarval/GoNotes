Bubble Tea · tea.Batch vs tea.Sequence ⭐⭐⭐☆

tea.Batch runs commands concurrently: results arrive in any order. tea.Sequence runs them one after another: results arrive in order. tea.Println prints a permanent line above the program's view.

// Bubble Tea example - tea.Batch vs tea.Sequence
//
// Example complexity rating: [★★★☆] 3/4
//
// tea.Batch runs commands concurrently: results arrive in any order.
// tea.Sequence runs them one after another: results arrive in order.
// tea.Println prints a permanent line above the program's view.
//
// Keys: b batch, s sequence, c clear, q quit
package main

import (
	"fmt"
	"os"
	"time"

	tea "charm.land/bubbletea/v2"
)

type doneMsg struct {
	name string
	took time.Duration
}

// job simulates work that takes d to finish.
func job(name string, d time.Duration) tea.Cmd {
	return func() tea.Msg {
		time.Sleep(d)
		return doneMsg{name, d}
	}
}

type model struct {
	log []string
}

func initialModel() model { return model{} }

func (m model) Init() tea.Cmd { return nil }

func (m model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case doneMsg:
		m.log = append(m.log, fmt.Sprintf("%-6s finished (%v)", msg.name, msg.took))
	case tea.KeyPressMsg:
		switch msg.String() {
		case "b": // all three start at once; "fast" finishes first
			m.log = append(m.log, "— batch —")
			return m, tea.Batch(
				job("slow", 900*time.Millisecond),
				job("medium", 500*time.Millisecond),
				job("fast", 100*time.Millisecond),
			)
		case "s": // each one waits for the previous one
			m.log = append(m.log, "— sequence —")
			return m, tea.Sequence(
				job("slow", 900*time.Millisecond),
				job("medium", 500*time.Millisecond),
				job("fast", 100*time.Millisecond),
				tea.Println("sequence complete ✔"), // printed above the UI
			)
		case "c":
			m.log = nil
		case "q", "ctrl+c":
			return m, tea.Quit
		}
	}
	return m, nil
}

func (m model) View() tea.View {
	s := "b batch • s sequence • c clear • q quit\n\n"
	for _, line := range m.log {
		s += line + "\n"
	}
	return tea.NewView(s)
}

func main() {
	if _, err := tea.NewProgram(initialModel()).Run(); err != nil {
		fmt.Println("error:", err)
		os.Exit(1)
	}
}
