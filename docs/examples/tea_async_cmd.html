Bubble Tea · async command (I/O without blocking the UI) ⭐⭐☆☆

A tea.Cmd is just func() tea.Msg. Bubble Tea runs it in its own goroutine, so slow work (HTTP, disk, database) never freezes the UI. The result comes back to Update as a message: success or error.

Keys: f fetch, q quit

go run . — 80×24
Lisbon: 18°C

f fetch again • q quit

Functions and types used (click to open in the cheatsheet):

tea.Cmd tea.Msg tea.Model tea.KeyPressMsg tea.Quit tea.View tea.NewView tea.NewProgram tea.Run

// Bubble Tea example - async command (I/O without blocking the UI)
//
// Example complexity rating: [★★☆☆] 2/4
//
// A tea.Cmd is just func() tea.Msg. Bubble Tea runs it in its own
// goroutine, so slow work (HTTP, disk, database) never freezes the UI.
// The result comes back to Update as a message: success or error.
//
// Keys: f fetch, q quit
package main

import (
	"errors"
	"fmt"
	"math/rand/v2"
	"os"
	"time"

	tea "charm.land/bubbletea/v2"
)

// Messages that carry the result of the work.
type weatherMsg struct{ celsius int }
type errMsg struct{ err error }

// fetchWeather simulates a slow network call. It does NOT touch the model:
// it only returns a message. Update decides what to do with it.
func fetchWeather(city string) tea.Cmd {
	return func() tea.Msg {
		time.Sleep(800 * time.Millisecond) // pretend this is the network
		if rand.IntN(4) == 0 {
			return errMsg{errors.New("connection reset by peer")}
		}
		return weatherMsg{celsius: 12 + rand.IntN(15)}
	}
}

type model struct {
	city    string
	loading bool
	temp    int
	err     error
}

func initialModel() model { return model{city: "Lisbon", loading: true} }

func (m model) Init() tea.Cmd {
	return fetchWeather(m.city) // fetch as soon as the program starts
}

func (m model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case weatherMsg:
		m.loading, m.err, m.temp = false, nil, msg.celsius
	case errMsg:
		m.loading, m.err = false, msg.err
	case tea.KeyPressMsg:
		switch msg.String() {
		case "f":
			if !m.loading {
				m.loading = true
				return m, fetchWeather(m.city)
			}
		case "q", "ctrl+c":
			return m, tea.Quit
		}
	}
	return m, nil
}

func (m model) View() tea.View {
	var s string
	switch {
	case m.loading:
		s = fmt.Sprintf("Fetching weather for %s...", m.city)
	case m.err != nil:
		s = fmt.Sprintf("Error: %v", m.err)
	default:
		s = fmt.Sprintf("%s: %d°C", m.city, m.temp)
	}
	return tea.NewView(s + "\n\nf fetch again • q quit\n")
}

func main() {
	if _, err := tea.NewProgram(initialModel()).Run(); err != nil {
		fmt.Println("error:", err)
		os.Exit(1)
	}
}
