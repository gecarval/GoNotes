// Bubbles example - animated progress bar
//
// Example complexity rating: [★★☆☆] 2/4
//
// progress.SetPercent returns a Cmd that animates the bar (spring
// physics) by sending progress.FrameMsg. You must forward FrameMsg to
// the bar's Update, or the animation freezes. Progress itself comes from
// a tea.Tick loop here.
//
// Keys: q quit
package main

import (
	"fmt"
	"os"
	"time"

	"charm.land/bubbles/v2/progress"
	tea "charm.land/bubbletea/v2"
)

type tickMsg time.Time

func tickCmd() tea.Cmd {
	return tea.Tick(500*time.Millisecond, func(t time.Time) tea.Msg { return tickMsg(t) })
}

type model struct {
	bar progress.Model
}

func initialModel() model {
	return model{bar: progress.New(progress.WithDefaultBlend(), progress.WithWidth(40))}
}

func (m model) Init() tea.Cmd { return tickCmd() }

func (m model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.KeyPressMsg:
		if s := msg.String(); s == "q" || s == "ctrl+c" {
			return m, tea.Quit
		}
	case tea.WindowSizeMsg:
		m.bar.SetWidth(min(msg.Width-4, 60))
	case tickMsg:
		if m.bar.Percent() >= 1.0 {
			return m, tea.Quit
		}
		// Two commands: the bar's animation and our next tick.
		return m, tea.Batch(m.bar.IncrPercent(0.1), tickCmd())
	case progress.FrameMsg:
		var cmd tea.Cmd
		m.bar, cmd = m.bar.Update(msg) // keeps the animation running
		return m, cmd
	}
	return m, nil
}

func (m model) View() tea.View {
	return tea.NewView("\n  Installing packages...\n\n  " + m.bar.View() + "\n\n  q quit\n")
}

func main() {
	if _, err := tea.NewProgram(initialModel()).Run(); err != nil {
		fmt.Println("error:", err)
		os.Exit(1)
	}
}
