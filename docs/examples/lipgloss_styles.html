Lip Gloss · basic styles ⭐☆☆☆

// Lip Gloss example - basic styles
//
// Example complexity rating: [★☆☆☆] 1/4
//
// A lipgloss.Style is an immutable value: every method returns a NEW
// style, so you can chain calls and derive variants from a base style.
// Render applies the style to a string and returns the ANSI result.
package main

import (
	"charm.land/lipgloss/v2"
)

func main() {
	// A base style...
	base := lipgloss.NewStyle().
		Padding(0, 2). // 0 rows top/bottom, 2 columns left/right
		MarginBottom(1)

	// ...and variants derived from it (base is not modified).
	title := base.
		Bold(true).
		Foreground(lipgloss.Color("#FAFAFA")).
		Background(lipgloss.Color("#7D56F4"))

	warning := base.
		Italic(true).
		Foreground(lipgloss.Color("#FFB86C"))

	ok := base.Foreground(lipgloss.Color("#50FA7B")).Underline(true)
	muted := base.Faint(true).Strikethrough(true)

	// lipgloss.Println downsamples colors to what the terminal supports.
	lipgloss.Println(title.Render("Lip Gloss"))
	lipgloss.Println(warning.Render("italic + orange"))
	lipgloss.Println(ok.Render("underlined + green"))
	lipgloss.Println(muted.Render("faint + strikethrough"))

	// Width + Align: the text is centered inside 30 cells.
	box := lipgloss.NewStyle().Width(30).Align(lipgloss.Center).Reverse(true)
	lipgloss.Println(box.Render("centered in 30 columns"))
}
