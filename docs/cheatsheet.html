type Model interface {
    Init() Cmd                         // First command to run (or nil)
    Update(Msg) (Model, Cmd)           // React to a message, return new state + next command
    View() View                        // Render the state (called after every Update)
}
type Msg = uv.Event                    // Any value can be a message
type Cmd func() Msg                    // Runs in a goroutine; its result is sent to Update
type BatchMsg []Cmd                    // What tea.Batch produces (handled by the runtime)

type View struct {
    Content         string             // The rendered screen (styled string)
    AltScreen       bool               // Full-screen alternate buffer
    MouseMode       MouseMode          // MouseModeNone | MouseModeCellMotion | MouseModeAllMotion
    Cursor          *Cursor            // Real terminal cursor (nil = hidden)
    WindowTitle     string             // Terminal window title
    ReportFocus     bool               // Receive FocusMsg / BlurMsg
    BackgroundColor color.Color        // Set the terminal background
    ForegroundColor color.Color        // Set the terminal foreground
    ProgressBar     *ProgressBar       // Native terminal progress bar (OSC 9;4)
